import requests

url = "http://127.0.0.1:5000/api/predict"
payload = {
    "age": 60,
    "sex": "Male",
    "chest_pain_type": "Typical angina",
    "resting_blood_pressure": 150,
    "cholestoral": 275,
    "fasting_blood_sugar": "Greater than 120 mg/ml",
    "rest_ecg": "ST-T wave abnormality",
    "Max_heart_rate": 125,
    "exercise_induced_angina": "Yes",
    "oldpeak": 2.4,
    "slope": "Flat",
    "vessels_colored_by_flourosopy": "Two",
    "thalassemia": "Reversable Defect"
}

response = requests.post(url, json=payload)
data = response.json()
print("Prediction:", data["prediction"])
print("Risk Probability:", data["probability"])
print("Consensus Agreement:", data["ensemble"]["consensus_status"])
